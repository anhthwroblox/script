--- CHỨC NĂNG ---

local correctKey = "anhthwroblox"
local getKeyLink = "https://taixuongmienphi.click/s/key-scriptz1u1lcqmlq4wqks"

-- Chức năng sao chép link
CopyLinkBtn.MouseButton1Click:Connect(function()
    setclipboard(getKeyLink)
    CopyLinkBtn.Text = "Đã chép link!"
    wait(2)
    CopyLinkBtn.Text = "Sao Chép Link Get Key"
end)

-- Chức năng kiểm tra key và chạy script
CheckKeyBtn.MouseButton1Click:Connect(function()
    if KeyInput.Text == correctKey then
        MainFrame:Destroy() -- Tắt bảng nhập key
        print("Key đúng! Đang khởi động Script...")
        
        -- Khởi động Script Blox Fruit của bạn
        loadstring(game:HttpGet("https://pastebin.com/raw/Ai5WqH8N", true))()
    else
        CheckKeyBtn.Text = "Sai Key! Thử lại"
        CheckKeyBtn.BackgroundColor3 = Color3.fromRGB(200, 0, 0)
        wait(2)
        CheckKeyBtn.Text = "Xác Nhận Key"
        CheckKeyBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 0)
    end
end)
