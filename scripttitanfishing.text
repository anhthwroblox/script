        wait(1)

        -- Chạy Script Blox Fruits tổng hợp của bạn
        loadstring(game:HttpGet("https://pastefy.app/hhiQgnJ2/raw"))()

        wait(2)
        CheckKeyBtn.Text = "XÁC NHẬN KEY"
